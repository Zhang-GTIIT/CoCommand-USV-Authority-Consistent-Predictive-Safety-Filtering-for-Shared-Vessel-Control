cmake_minimum_required(VERSION 3.20)
project(CoCommandC1Bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RPLIDAR_SDK_ROOT "" CACHE PATH "Local official Slamtec/rplidar_sdk checkout")
if(NOT EXISTS "${RPLIDAR_SDK_ROOT}/sdk/include/sl_lidar.h")
  message(FATAL_ERROR "Set RPLIDAR_SDK_ROOT to an existing SDK checkout; no configure-time network downloads")
endif()
set(SDK "${RPLIDAR_SDK_ROOT}/sdk")
if(WIN32)
  set(ARCH win32)
elseif(APPLE)
  set(ARCH macOS)
else()
  set(ARCH linux)
endif()
file(GLOB SDK_CORE CONFIGURE_DEPENDS "${SDK}/src/sl_*.cpp")
file(GLOB_RECURSE SDK_UNPACK CONFIGURE_DEPENDS "${SDK}/src/dataunpacker/*.cpp")
add_library(rplidar_sdk STATIC ${SDK_CORE} ${SDK_UNPACK} "${SDK}/src/hal/thread.cpp"
  "${SDK}/src/arch/${ARCH}/net_serial.cpp" "${SDK}/src/arch/${ARCH}/net_socket.cpp"
  "${SDK}/src/arch/${ARCH}/timer.cpp")
target_include_directories(rplidar_sdk PUBLIC "${SDK}/include" PRIVATE "${SDK}/src"
  "${SDK}/src/dataunpacker" "${SDK}/src/dataunpacker/unpacker")
if(WIN32)
  target_compile_definitions(rplidar_sdk PRIVATE _WIN32_WINNT=0x0601 NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(rplidar_sdk PUBLIC ws2_32 winmm)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(rplidar_sdk PUBLIC Threads::Threads)
endif()
add_executable(cocommand_c1 c1_bridge.cpp)
target_link_libraries(cocommand_c1 PRIVATE rplidar_sdk)
if(MSVC)
  target_compile_options(rplidar_sdk PRIVATE /W3 /EHsc)
  target_compile_options(cocommand_c1 PRIVATE /W4 /EHsc)
else()
  target_compile_options(rplidar_sdk PRIVATE -Wall -Wextra)
  target_compile_options(cocommand_c1 PRIVATE -Wall -Wextra -Wpedantic)
endif()
configure_file("${RPLIDAR_SDK_ROOT}/LICENSE" "${CMAKE_BINARY_DIR}/SLAMTEC-SDK-LICENSE.txt" COPYONLY)
